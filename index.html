-- ========================================================= -- DEMO COIN PURCHASE -- ========================================================= CREATE OR REPLACE FUNCTION public.add_demo_coins( p_amount bigint ) RETURNS bigint LANGUAGE plpgsql SECURITY DEFINER SET search_path = public AS $$ DECLARE new_balance bigint; BEGIN IF auth.uid() IS NULL THEN RAISE EXCEPTION 'Login required'; END IF; IF p_amount <= 0 OR p_amount > 100000 THEN RAISE EXCEPTION 'Invalid coin amount'; END IF; INSERT INTO public.wallets ( user_id, coins ) VALUES ( auth.uid(), p_amount ) ON CONFLICT (user_id) DO UPDATE SET coins = public.wallets.coins + EXCLUDED.coins, updated_at = now(); SELECT coins INTO new_balance FROM public.wallets WHERE user_id = auth.uid(); RETURN new_balance; END; $$; GRANT EXECUTE ON FUNCTION public.add_demo_coins(bigint) TO authenticated; -- ========================================================= -- LIVE STREAMS -- ========================================================= ALTER TABLE public.live_streams ENABLE ROW LEVEL SECURITY; -- ========================================================= -- SEND GIFT -- Coins are deducted server-side. -- ========================================================= CREATE OR REPLACE FUNCTION public.send_gift( p_live_id uuid, p_points integer ) RETURNS bigint LANGUAGE plpgsql SECURITY DEFINER SET search_path = public AS $$ DECLARE host_user uuid; current_coins bigint; new_balance bigint; BEGIN IF auth.uid() IS NULL THEN RAISE EXCEPTION 'Login required'; END IF; IF p_points <= 0 OR p_points > 10000 THEN RAISE EXCEPTION 'Invalid gift'; END IF; SELECT host_id INTO host_user FROM public.live_streams WHERE id = p_live_id AND status = 'live' FOR UPDATE; IF host_user IS NULL THEN RAISE EXCEPTION 'Live is not active'; END IF; INSERT INTO public.wallets(user_id,coins) VALUES(auth.uid(),0) ON CONFLICT (user_id) DO NOTHING; SELECT coins INTO current_coins FROM public.wallets WHERE user_id = auth.uid() FOR UPDATE; IF current_coins < p_points THEN RAISE EXCEPTION 'Not enough coins'; END IF; UPDATE public.wallets SET coins = coins - p_points, updated_at = now() WHERE user_id = auth.uid() RETURNING coins INTO new_balance; INSERT INTO public.live_gifts ( live_id, sender_id, receiver_id, points ) VALUES ( p_live_id, auth.uid(), host_user, p_points ); RETURN new_balance; END; $$; GRANT EXECUTE ON FUNCTION public.send_gift(uuid,integer) TO authenticated;
